Elephant Detection System - Technical Details

Designed to minimize human–wildlife conflict in rural/border zones by analyzing video streams using custom YOLOv8 model and broadcasting alerts. 

import cv2
from ultralytics import YOLO
import requests

# Load custom YOLOv8 model for elephant detection
model = YOLO('models/elephant_yolov8.pt')
cap = cv2.VideoCapture(0)  # Camera feed stream

while cap.isOpened():
    ret, frame = cap.read()
    if not ret: break

    results = model(frame, conf=0.75)
    for r in results:
        for box in r.boxes:
            if int(box.cls) == 20:  # Elephant Class ID
                x1, y1, x2, y2 = map(int, box.xyxy[0])
                cv2.rectangle(frame, (x1, y1), (x2, y2), (0, 255, 0), 2)
                
                # Dispatch alert to IoT siren gateway
                requests.post("http://iot-gateway.local/alert", json={"zone": "Forest_B4", "alert": "ELEPHANT"})

    cv2.imshow("Detection Feed", frame)
    if cv2.waitKey(1) & 0xFF == ord('q'): break

Notes-to-Story: AI Video Generation Pipeline

An asynchronous AI engine converting raw document notes into animated educational videos with character narrations, script generation, and SQLite job state tracking. 

import asyncio
import json
import sqlite3
from app.services.video_renderer import generate_scene_video
from app.services.script_generator import generate_heuristic_script

async def render_notes_to_story(job_id: str):
    """Async pipeline for rendering Notes-to-Story AI video scenes."""
    conn = sqlite3.connect("backend/notes_to_story.db")
    c = conn.cursor()
    c.execute("SELECT summary_json FROM job_states WHERE id = ?", (job_id,))
    row = c.fetchone()
    
    summary_data = json.loads(row[0]) if row else {}
    # Generate NLP script with character actions & scene beats
    script_data = generate_heuristic_script(summary_data)
    
    # Render scene video with audio TTS & animated frames
    video_url = await generate_scene_video(script_data, job_id)
    
    c.execute("""
        UPDATE job_states
        SET script_json = ?, video_url = ?, status = 'completed', progress_percentage = 100.0
        WHERE id = ?
    """, (json.dumps(script_data), video_url, job_id))
    conn.commit()
    conn.close()
    return video_url
